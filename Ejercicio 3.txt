    <script>
        let personas = [];


        function serializar() {
            const nombre = document.getElementById("inputNombre").value;
            const apellidos = document.getElementById("inputApellidos").value;
            const edad = document.getElementById("inputEdad").value;
            const ciudad = document.getElementById("inputCiudad").value;
           
            const hobbies = Array.from(document.querySelectorAll('input[name="hobbies"]:checked'))
                                .map(hobby => hobby.value);


            if (!nombre) return alert("El campo Nombre es obligatorio");
            if (!apellidos) return alert("El campo Apellidos es obligatorio");
            if (!edad) return alert("El campo Edad es obligatorio");
            if (!ciudad) return alert("El campo Ciudad es obligatorio");
            if (hobbies.length === 0) return alert("Debe seleccionar al menos un hobby");


            const persona = { id: new Date().getTime(), nombre, apellidos, edad: parseInt(edad), ciudad, hobbies };
            personas.push(persona);
            actualizarTabla();
        }


        function actualizarTabla() {
            const json = JSON.stringify(personas);
            document.getElementById("jsonOutput").textContent = json;
            generaTabla(json);
        }


        function generaTabla(json) {
            const datos = JSON.parse(json);
            const tbody = document.getElementById("tablaBody");
            tbody.innerHTML = "";
            datos.forEach(persona => {
                tbody.innerHTML += generaTR(persona);
            });
            actualizarEstadisticas();
        }


        function generaTR(persona) {
            return `<tr id="fila-${persona.id}" onclick="eliminarFila(${persona.id})">
                        <td>${persona.nombre}</td>
                        <td>${persona.apellidos}</td>
                        <td>${persona.edad}</td>
                        <td>${persona.ciudad}</td>
                        <td>${persona.hobbies.join(", ")}</td>
                    </tr>`;
        }


        function eliminarFila(id) {
            document.getElementById(`fila-${id}`).remove();
            const index = personas.findIndex(p => p.id === id);
            if (index !== -1) {
                personas.splice(index, 1);
            }
            actualizarTabla();
        }


        function actualizarEstadisticas() {
            if (personas.length === 0) {
                document.getElementById("stats").textContent = "No hay datos disponibles";
                return;
            }
            const edades = personas.map(p => p.edad);
            const suma = edades.reduce((a, b) => a + b, 0);
            const media = (suma / edades.length).toFixed(2);
            const max = Math.max(...edades);
            const min = Math.min(...edades);
            document.getElementById("stats").textContent = `Suma: ${suma}, Media: ${media}, Máximo: ${max}, Mínimo: ${min}`;
            console.log("Estadísticas - Suma:", suma, "Media:", media, "Máximo:", max, "Mínimo:", min);
        }
    </script>
